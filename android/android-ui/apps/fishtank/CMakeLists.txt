if(NOT WINDOWS_MSVC_X86_64)
  # On Linux and macOS, we avoid direct linking of SwiftShader libraries
  # because some prebuilts have malformed symbol tables that cause LLD to
  # fail. Instead, we rely on RPATH-based resolution at runtime. This avoids
  # needing to set LD_LIBRARY_PATH or DYLD_LIBRARY_PATH, which can be
  # restricted by security features like macOS System Integrity Protection (SIP).
  # Since fishtank uses LazyLoadedEGLDispatch to load these dynamically
  # at runtime, direct linking is not required.
  android_target_dependency(fishtank all SWIFTSHADER_DEPENDENCIES)

  if(NOT APPLE)
    # Linux-specific RPATH (Apple RPATH is set in the block above)
    android_target_properties(
      fishtank all "INSTALL_RPATH>=\$ORIGIN/lib64/gles_swiftshader")
  endif()
endif()

# Strips the given prebuilt executable during install.
# We define this locally because android_strip_prebuilt in android.cmake
# is currently broken for standalone distributions (it evaluates CMAKE_INSTALL_PREFIX
# at configuration time instead of install time).
# Strips and signs the given file during install.
# We handle both in one function to ensure the correct order on macOS:
# Stripping must happen BEFORE signing, or the signature will be invalidated.
function(fishtank_post_install FNAME)
  if(NOT WINDOWS_MSVC_X86_64)
    # 1. Strip the binary if requested
    install(
      CODE "if(CMAKE_INSTALL_DO_STRIP AND OPTION_FISHTANK_STRIP)
              execute_process(COMMAND \"${CMAKE_STRIP}\" \"$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/${FNAME}\")
            endif()"
      COMPONENT fishtank
      EXCLUDE_FROM_ALL)

    # 2. Sign the binary on macOS (always, even if not stripped)
    if(APPLE)
      install(
        CODE "message(STATUS \"Signing   : ${FNAME}\")
              execute_process(COMMAND codesign -s - --entitlements ${ANDROID_QEMU2_TOP_DIR}/entitlements.plist \"$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/${FNAME}\")"
        COMPONENT fishtank
        EXCLUDE_FROM_ALL)
    endif()
  endif()
endfunction()

# Helper function to install dependencies with platform-specific destination logic.
#
# This function handles the distribution of various dependency types to the standalone
# fishtank directory. It supports:
#   1. CMake Targets: Shared libraries are installed to lib64/ on Unix or the root on Windows.
#   2. Mapping Strings: Strings in the format "source/path>dest/rel/path" are parsed
#      and installed. On Windows, the destination is flattened to the root.
#
# Arguments:
#   DEPS: A list of CMake targets or "src>dst" mapping strings.
function(fishtank_install_dependency)
  set(options)
  set(oneValueArgs)
  set(multiValueArgs DEPS)
  cmake_parse_arguments(INST "${options}" "${oneValueArgs}" "${multiValueArgs}"
                        ${ARGN})

  foreach(dep ${INST_DEPS})
    if(dep MATCHES ".*>.*")
      # Handle "src>dst" mapping
      string(REPLACE ">" ";" SRC_DST ${dep})
      list(GET SRC_DST 0 SRC)
      list(GET SRC_DST 1 DST)
      get_filename_component(DST_DIR ${DST} DIRECTORY)
      get_filename_component(DST_NAME ${DST} NAME)
      get_filename_component(REAL_SRC ${SRC} REALPATH)
      if(WINDOWS_MSVC_X86_64)
        # On Windows, we typically flatten DLLs to the root so the OS loader
        # can find them. We identify these because they are usually mapped
        # to lib64/. However, some dependencies (like SwiftShader or Angle)
        # must remain in their subdirectories.
        if(DST_DIR STREQUAL "lib64")
          set(FISHTANK_DEST "fishtank")
        else()
          set(FISHTANK_DEST "fishtank/${DST_DIR}")
        endif()
        install(
          FILES ${REAL_SRC}
          DESTINATION ${FISHTANK_DEST}
          RENAME ${DST_NAME}
          COMPONENT fishtank
          EXCLUDE_FROM_ALL)
      else()
        install(
          FILES ${REAL_SRC}
          DESTINATION fishtank/${DST_DIR}
          RENAME ${DST_NAME}
          COMPONENT fishtank
          EXCLUDE_FROM_ALL)
        if(DST_NAME MATCHES ".*\\.so.*" OR DST_NAME MATCHES ".*\\.dylib")
          fishtank_post_install("fishtank/${DST}")
        endif()
        # Extra logic for libc++ on Linux to support QtWebEngineProcess.
        if(LINUX_X86_64 AND (DST_NAME STREQUAL "libc++.so.1"
                             OR DST_NAME STREQUAL "libc++.so"))
          install(
            FILES ${REAL_SRC}
            DESTINATION fishtank/lib64/qt/lib
            RENAME ${DST_NAME}
            COMPONENT fishtank
            EXCLUDE_FROM_ALL)
          fishtank_post_install("fishtank/lib64/qt/lib/${DST_NAME}")
        endif()
      endif()
    elseif(TARGET ${dep})
      # Handle CMake targets
      if(WINDOWS_MSVC_X86_64)
        install(
          FILES $<TARGET_FILE:${dep}>
          DESTINATION fishtank
          COMPONENT fishtank
          EXCLUDE_FROM_ALL)
      else()
        install(
          TARGETS ${dep}
          RUNTIME DESTINATION fishtank/lib64
          LIBRARY DESTINATION fishtank/lib64
          COMPONENT fishtank
          EXCLUDE_FROM_ALL)
        if(APPLE)
          if(dep MATCHES "^lib.*")
            fishtank_post_install("fishtank/lib64/${dep}${CMAKE_SHARED_LIBRARY_SUFFIX}")
          else()
            fishtank_post_install("fishtank/lib64/lib${dep}${CMAKE_SHARED_LIBRARY_SUFFIX}")
          endif()
        endif()
        # We don't need android_strip_prebuilt here because targets are stripped
        # automatically by cmake --install --strip (which we added to compile.py)
      endif()
    endif()
  endforeach()
endfunction()

# dependencies are NOT included in the default emulator distribution.

# directory (objs/distribution-fishtank) during the specialized build phase.

  EXCLUDE_FROM_ALL)

# For Qt dependencies, the only special thing we do in here is, for windows, flatten all dlls in
# qt/lib to the root. Everything else is installed as described in emu-qt5-config.cmake.
foreach(qt_dep ${QT5_SHARED_DEPENDENCIES} ${QT5_WEBENGINE_SHARED_DEPENDENCIES})
  if(qt_dep MATCHES ".*>>.*")
    string(REPLACE ">>" ";" SRC_DST ${qt_dep})
    list(GET SRC_DST 0 SRC)
    list(GET SRC_DST 1 DST)
    if(SRC MATCHES ".*\\*.*")
      file(GLOB GLOBBED_FILES ${SRC})
      install(
        FILES ${GLOBBED_FILES}
        DESTINATION fishtank/${DST}
        COMPONENT fishtank
        EXCLUDE_FROM_ALL)
    else()
      install(
        DIRECTORY ${SRC}
        DESTINATION fishtank/${DST}
        COMPONENT fishtank
        EXCLUDE_FROM_ALL)
    endif()
  else()
    string(REPLACE ">" ";" SRC_DST ${qt_dep})
    list(GET SRC_DST 0 SRC)
    list(GET SRC_DST 1 DST)
    get_filename_component(DST_DIR ${DST} DIRECTORY)
    if(WINDOWS_MSVC_X86_64)
      # On Windows, the DLLs must be next to the executable.
      # Plugins and resources can stay in their subdirectories,
      # but core DLLs like Qt6Core must be findable by the OS loader.
      # We can identify core DLLs because they are installed in lib64/qt/lib.
      if(SRC MATCHES ".*\\.dll$" AND DST MATCHES "lib64/qt/lib/.*")
        install(
          FILES ${SRC}
          DESTINATION fishtank
          COMPONENT fishtank
          EXCLUDE_FROM_ALL)
      else()
        install(
          FILES ${SRC}
          DESTINATION fishtank/${DST_DIR}
          COMPONENT fishtank
          EXCLUDE_FROM_ALL)
      endif()
    else()
      # On Linux/MacOS, the Qt installation rules defined in emu-qt5-config.cmake don't need any
      # modification.
      if(SRC MATCHES "QtWebEngineProcess$")
        install(
          PROGRAMS ${SRC}
          DESTINATION fishtank/${DST_DIR}
          COMPONENT fishtank
          EXCLUDE_FROM_ALL)
        fishtank_post_install("fishtank/${DST}")
      else()
        install(
          FILES ${SRC}
          DESTINATION fishtank/${DST_DIR}
          COMPONENT fishtank
          EXCLUDE_FROM_ALL)
        if(SRC MATCHES ".*\\.so.*" OR SRC MATCHES ".*\\.dylib")
          fishtank_post_install("fishtank/${DST}")
        endif()
      endif()
    endif()
  endif()
endforeach()
